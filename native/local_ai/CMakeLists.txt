cmake_minimum_required(VERSION 3.22)
project(antisocial_local_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANTISOCIAL_QVAC_SOURCE)
    message(FATAL_ERROR "Use scripts/build_local_ai.py to acquire the pinned runtime")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory("${ANTISOCIAL_QVAC_SOURCE}" qvac)
add_library(antisocial_text_core STATIC text_runtime.cpp)
target_include_directories(antisocial_text_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(antisocial_text_core PRIVATE "${ANTISOCIAL_QVAC_SOURCE}/src")
target_link_libraries(antisocial_text_core PUBLIC llama llama-common)
target_compile_definitions(antisocial_text_core PRIVATE ANTISOCIAL_QVAC_REVISION="bc47470138d892a45fedb037c2615cc12c54cded+prism-v2-11809184898113482c2f510fd13da669aaba1052+antisocial-kv-v3")
if(ANDROID)
    add_library(antisocial_local_ai SHARED jni.cpp)
    target_link_libraries(antisocial_local_ai PRIVATE antisocial_text_core log)
    target_link_options(antisocial_local_ai PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(antisocial-text-reference reference.cpp)
    target_link_libraries(antisocial-text-reference PRIVATE antisocial_text_core)
    add_executable(antisocial-kv-quality-reference kv_quality_reference.cpp)
    target_link_libraries(antisocial-kv-quality-reference PRIVATE antisocial_text_core)
    add_executable(antisocial-kv-speculative-reference kv_speculative_reference.cpp)
    target_include_directories(antisocial-kv-speculative-reference PRIVATE "${ANTISOCIAL_QVAC_SOURCE}/src")
    target_link_libraries(antisocial-kv-speculative-reference PRIVATE llama llama-common)
    add_executable(antisocial-text-kernel-test kernel_reference.cpp)
    target_include_directories(antisocial-text-kernel-test PRIVATE "${ANTISOCIAL_QVAC_SOURCE}/ggml/src")
    target_link_libraries(antisocial-text-kernel-test PRIVATE llama)
    foreach(probe IN ITEMS attention basis logits calibrate state position)
        add_executable(antisocial-kv-${probe}-reference kv_${probe}_reference.cpp)
        target_include_directories(antisocial-kv-${probe}-reference PRIVATE "${ANTISOCIAL_QVAC_SOURCE}/src" "${ANTISOCIAL_QVAC_SOURCE}/ggml/src")
        target_link_libraries(antisocial-kv-${probe}-reference PRIVATE llama llama-common)
    endforeach()
endif()

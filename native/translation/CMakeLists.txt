cmake_minimum_required(VERSION 3.22)
project(antisocial_translation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GIT_SUBMODULE OFF CACHE BOOL "Sources are acquired and verified by the build script" FORCE)
set(COMPILE_CPU ON CACHE BOOL "" FORCE)
set(COMPILE_CUDA OFF CACHE BOOL "" FORCE)
set(COMPILE_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(COMPILE_TESTS OFF CACHE BOOL "" FORCE)
set(COMPILE_LIBRARY_ONLY ON CACHE BOOL "" FORCE)
set(USE_WASM_COMPATIBLE_SOURCE ON CACHE BOOL "Minimal inference only" FORCE)
set(COMPILE_WITHOUT_EXCEPTIONS OFF CACHE BOOL "" FORCE)
set(USE_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(USE_MKL OFF CACHE BOOL "" FORCE)
set(USE_FBGEMM OFF CACHE BOOL "" FORCE)
set(USE_NCCL OFF CACHE BOOL "" FORCE)
set(SSPLIT_USE_INTERNAL_PCRE2 OFF CACHE BOOL "Pinned PCRE2 is built first" FORCE)
set(SSPLIT_COMPILE_LIBRARY_ONLY ON CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(USE_THREADS OFF CACHE BOOL "Single serial translation worker" FORCE)
add_compile_options(-ffunction-sections -fdata-sections)
add_compile_definitions(COMPILE_CPU=1 USE_SENTENCEPIECE _USE_INTERNAL_STRING_VIEW)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|armv8)")
  set(BUILD_ARCH armv8-a CACHE STRING "" FORCE)
  set(USE_ONNX_SGEMM OFF CACHE BOOL "ARM uses RUY" FORCE)
  add_compile_definitions(ARM FMA SSE COMPILE_CPU=1 USE_SENTENCEPIECE _USE_INTERNAL_STRING_VIEW)
  add_compile_options(-flax-vector-conversions -fno-strict-aliasing)
else()
  set(BUILD_ARCH nehalem CACHE STRING "SSE4.1 baseline, no host AVX assumptions" FORCE)
  set(USE_ONNX_SGEMM ON CACHE BOOL "Portable CPU SGEMM" FORCE)
endif()
if(ANDROID)
  set(CMAKE_THREAD_LIBS_INIT "")
  set(CMAKE_HAVE_THREADS_LIBRARY 1)
  set(CMAKE_USE_PTHREADS_INIT 1)
  set(THREADS_PREFER_PTHREAD_FLAG OFF)
  set(Threads_FOUND TRUE)
  add_compile_options(-fvisibility=hidden $<$<COMPILE_LANGUAGE:CXX>:-fvisibility-inlines-hidden> -fstack-protector-strong)
  add_compile_definitions(_FORTIFY_SOURCE=2)
endif()
add_subdirectory(${ANTISOCIAL_TRANSLATIONS_SOURCE}/inference bergamot EXCLUDE_FROM_ALL)
# Old Marian promotes every new compiler diagnostic to a build error. Keep
# diagnostics visible without forcing consumers to freeze their compiler.
get_target_property(marian_warnings marian COMPILE_OPTIONS)
list(REMOVE_ITEM marian_warnings -Werror)
set_property(TARGET marian PROPERTY COMPILE_OPTIONS ${marian_warnings})
file(GLOB fasttext_sources CONFIGURE_DEPENDS "${ANTISOCIAL_FASTTEXT_SOURCE}/src/*.cc")
list(FILTER fasttext_sources EXCLUDE REGEX "/main\\.cc$")
add_library(antisocial_fasttext STATIC ${fasttext_sources})
target_include_directories(antisocial_fasttext PUBLIC ${ANTISOCIAL_FASTTEXT_SOURCE}/src)
target_compile_options(antisocial_fasttext PRIVATE -O3)
if(ANDROID)
  add_library(antisocial_translation SHARED translation.cpp)
  target_compile_definitions(antisocial_translation PRIVATE ANTISOCIAL_ANDROID=1)
  target_link_libraries(antisocial_translation PRIVATE bergamot-translator-source antisocial_fasttext log)
  target_link_options(antisocial_translation PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
else()
  add_executable(antisocial-translation-reference translation.cpp)
  target_link_libraries(antisocial-translation-reference PRIVATE bergamot-translator-source antisocial_fasttext)
endif()
